// Store the tasks in a global variable
let taskList = [];

// Add new task
function addTask() {
  const taskInput = document.getElementById('taskName');
  const dueDateInput = document.getElementById('dueDate');
  const taskName = taskInput.value.trim();
  const dueDate = dueDateInput.value;

  if (taskName !== "") {
    // Add the task to the task list
    const task = {
      name: taskName,
      dueDate: dueDate,
      isDone: false,
    };
    taskList.push(task);

    taskInput.value = ""; // Clear the input field
    dueDateInput.value = ""; // Clear the date input field

    renderTasks(); // Update the UI
    saveTasks(); // Save to localStorage
  }
}

// Toggle task completion
function toggleTaskCompletion(index) {
  taskList[index].isDone = !taskList[index].isDone; // Toggle completion status
  renderTasks(); // Re-render tasks
  saveTasks(); // Save to localStorage
}

// Delete task
function deleteTask(index) {
  taskList.splice(index, 1); // Remove task
  renderTasks(); // Re-render tasks
  saveTasks(); // Save to localStorage
}

// Render tasks on the page
function renderTasks() {
  const taskListElement = document.getElementById('tasks');
  taskListElement.innerHTML = ""; // Clear the task list

  taskList.forEach((task, index) => {
    const li = document.createElement('li');
    li.className = task.isDone ? "completed" : ""; // Add 'completed' class if done
    li.innerHTML = `${task.name} - Due: ${task.dueDate}
      <button onclick="toggleTaskCompletion(${index})">Complete</button>
      <button onclick="deleteTask(${index})">Delete</button>`;
    taskListElement.appendChild(li);
  });
}

// Save tasks to localStorage
function saveTasks() {
  localStorage.setItem('tasks', JSON.stringify(taskList));
}

// Load tasks from localStorage when the page loads
function loadTasks() {
  const savedTasks = localStorage.getItem('tasks');
  if (savedTasks) {
    taskList = JSON.parse(savedTasks);
    renderTasks();
  }
}

// Load tasks when the page is loaded
window.onload = loadTasks;
